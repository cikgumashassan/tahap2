        // START QUIZ
        function startQuiz() {
            const name = document.getElementById('student-name').value.trim();
            const year = document.getElementById('student-year').value;
            const topic = document.getElementById('quiz-topic').value;

            if (!name) return;

            currentStudent = { name, year, topic };

            // Filter questions
            let filtered = questionBank.filter(q => q.years.includes(year));
            if (topic !== 'all') {
                filtered = filtered.filter(q => q.topic === topic);
            }

            // Shuffle questions
            activeQuestions = [...filtered].sort(() => 0.5 - Math.random()).slice(0, 10);

            if (activeQuestions.length === 0) {
                alert('Tiada soalan ditemui untuk kriteria ini. Sila pilih tajuk lain.');
                return;
            }

            // Reset States
            currentQIndex = 0;
            score = 0;

            // Update UI Badges
            document.getElementById('quiz-year-badge').innerText = year;
            document.getElementById('quiz-topic-badge').innerText = topic.toUpperCase();
            document.getElementById('total-q-num').innerText = activeQuestions.length;
            document.getElementById('current-score').innerText = '0';

            showScreen('screen-quiz');
